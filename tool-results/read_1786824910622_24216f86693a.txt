  1008→/**
  1009→ * Standard play-family structure derived from a single base concept.
  1010→ * One blocking framework (e.g. Inside Zone) generates an entire family.

  1012→export interface PlayFamilyTemplate {
  1013→  /** Base concept id, e.g. "inside_zone". */
  1014→  base: string;
  1015→  /** Read concept built on the base (e.g. Zone Read). */
  1016→  read: string;
  1017→  /** RPO off the base (e.g. IZ Glance). */
  1018→  rpo: string;
  1019→  /** Constraint play that punishes defense for cheating (e.g. Jet Sweep). */
  1020→  constraint: string;
  1021→  /** Counter play that punishes over-pursuit (e.g. Counter Bash). */
  1022→  counter: string;
  1023→  /** Play-action off the base (e.g. IZ PA Flood). */
  1024→  playAction: string;
  1025→  /** Shot play off the base (e.g. Glance-and-Go). */
  1026→  shot: string;
  1027→  /** Screen off the base fake (e.g. Fake IZ RB Screen). */
  1028→  screen: string;
  1029→  /** Boot / movement pass off the base action. */
  1030→  boot: string;
  1031→}

  1039→ * buildPlayFamily("inside_zone")
